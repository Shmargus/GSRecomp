# Golden Sun: The Lost Age Recompiled.
#
# The engine (gbarecomp), the bare runner (src/tla_runner.cpp), the translated
# game code from scripts/tla_recompile.ps1, and the player builder.
#
# Built like Golden Sun Recompiled's speed build
# (../scripts/build_game.ps1 -Game LostAge): the engine is TLAEngine.dll (run by
# TLARecomp.exe; Linux: the TLARecomp program itself), link-time
# optimised; the translated game code is TLAGame.dll beside it, compiled
# without LTO; and the game code is the block-timing copy
# (scripts/block_timing.py).

set(CMAKE_CXX_STANDARD 20)

# Link-time optimisation of TLARecomp.exe and the gbarecomp libraries. In a
# split build that is the engine only. Degrades to no LTO (with a message
# build_game.ps1 stops on) if the toolchain cannot do it.
option(TLA_ENABLE_LTO "Link-time optimise TLARecomp and gbarecomp" ON)
set(TLA_IPO_SUPPORTED FALSE)

    include(CheckIPOSupported)
    check_ipo_supported(RESULT TLA_IPO_SUPPORTED OUTPUT TLA_IPO_SUPPORT_ERROR)
    if(NOT TLA_IPO_SUPPORTED)
        message(STATUS "TLA_ENABLE_LTO is ON but unsupported "
            "(${TLA_IPO_SUPPORT_ERROR}); building without it.")
    endif()
endif()

# gbarecomp's own libraries read Golden Sun 1's names for the same switch.

set(GSR_IPO_SUPPORTED ${TLA_IPO_SUPPORTED})

# TLA_SPLIT_GAME_CODE: the engine (our code only) is TLARecomp.exe and the
# translated game code is TLAGame.dll beside it, as in Golden Sun
# Recompiled, so a release can ship the engine prebuilt and the player's PC
# builds only the DLL from their ROM. Windows joins the two when the game
# starts: the DLL exports the tables the engine reads (TLAGame.def, written
# below) and the exe exports everything the game code calls. OFF builds one
# exe.
option(TLA_SPLIT_GAME_CODE "Build translated game code as a separate library" ON)
if(TLA_SPLIT_GAME_CODE AND NOT (MINGW OR CMAKE_SYSTEM_NAME STREQUAL "Linux"))
    message(FATAL_ERROR "TLA_SPLIT_GAME_CODE requires MinGW or Linux.")
endif()
if(TLA_SPLIT_GAME_CODE)
    # Both platforms compile the player's code separately from the LTO engine.
    add_library(tla_game_objects OBJECT ${TLA_GAME_SOURCES})
    set_target_properties(tla_game_objects PROPERTIES
        INTERPROCEDURAL_OPTIMIZATION FALSE)
    set(TLA_GAME_CODE_OPT "-Og" CACHE STRING
        "Optimisation flag for the game code in a split build")
    target_compile_options(tla_game_objects PRIVATE ${TLA_GAME_CODE_OPT} -g0)
    target_include_directories(tla_game_objects PRIVATE ${TLA_GAME_INCLUDE_DIRS})
    # How many game-code files compile at once (0 = no limit). One file can
    # need 6.4 GiB (reloc/harv60, -Og, measured 2026-10-16); fourteen at once
    # ran a 16 GB machine out of memory. scripts/build_game.ps1 sets it from
    # the build's memory limit; the rest of the build keeps its full count.
    set(TLA_GAME_CODE_COMPILE_JOBS 0 CACHE STRING
        "Game-code files compiled at once (0 = no limit)")
    if(TLA_GAME_CODE_COMPILE_JOBS GREATER 0)
        set_property(GLOBAL APPEND PROPERTY JOB_POOLS
            tla_game_code=${TLA_GAME_CODE_COMPILE_JOBS})
        set_property(TARGET tla_game_objects PROPERTY
            JOB_POOL_COMPILE tla_game_code)
    endif()
endif()

if(TLA_SPLIT_GAME_CODE AND MINGW)
    # Windows: the engine is a plug-in, TLAEngine.dll (target TLARecomp, so
    # the per-file object paths stay the same), run by TLARecomp.exe, the
    # small player program beside it (engine player_main.cpp). One window
    # for both games (STATUS.md, "One program for both games").
    add_library(TLARecomp SHARED ${TLA_ENGINE_SOURCES}
        "${GBARECOMP_ROOT}/src/runtime/game_module.cpp")
    set_target_properties(TLARecomp PROPERTIES
        PREFIX "" OUTPUT_NAME TLAEngine
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    add_executable(TLAPlayer "${GBARECOMP_ROOT}/src/runtime/player_main.cpp")
    set_target_properties(TLAPlayer PROPERTIES OUTPUT_NAME TLARecomp)
    target_compile_definitions(TLAPlayer PRIVATE GSR_ENGINE_PLUGIN=L"TLAEngine.dll")
    # The game runs on the player program's threads.
    target_link_options(TLAPlayer PRIVATE -Wl,--stack,16777216)
    target_link_libraries(TLAPlayer PRIVATE shell32)
    add_dependencies(TLAPlayer TLARecomp TLAGame)
elseif(TLA_SPLIT_GAME_CODE)
    add_executable(TLARecomp ${TLA_ENGINE_SOURCES})
else()
    add_executable(TLARecomp ${TLA_ENGINE_SOURCES} ${TLA_GAME_SOURCES})
endif()
if(TLA_IPO_SUPPORTED)
    set_property(TARGET TLARecomp PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
endif()

# Generated game code calls one shared copy of the RAM/ROM access fast paths
# instead of inlining them at every load and store (runtime_arm.h,
# GBARECOMP_OUTLINE_BUS). Same behaviour; much less code to compile.
set(TLA_GAME_CODE_SOURCES ${TLA_GAME_SOURCES})
list(FILTER TLA_GAME_CODE_SOURCES INCLUDE REGEX "/recompiled[^/]*\\.cpp$")
set_property(SOURCE ${TLA_GAME_CODE_SOURCES} APPEND PROPERTY
    COMPILE_DEFINITIONS GBARECOMP_OUTLINE_BUS=1)

if(WIN32)
    target_link_libraries(TLARecomp PRIVATE shell32)
endif()

target_include_directories(TLARecomp PRIVATE ${TLA_GAME_INCLUDE_DIRS})
# Only the unpacker guard reaches the player builder's header; The Broken
# Seal's src/ is never on TLA's include path (same-named headers).
set_property(SOURCE "${CMAKE_SOURCE_DIR}/../The Broken Seal/src/unpacker_guard.cpp"
    APPEND PROPERTY INCLUDE_DIRECTORIES "${GSR_PLAYER_BUILDER_DIR}")
if(TLA_PLAYER_RELEASE)
    target_compile_definitions(TLARecomp PRIVATE TLA_PLAYER_RELEASE=1)
endif()
target_compile_definitions(TLARecomp PRIVATE
    GBARECOMP_DEFAULT_DEBUG_PORT=19843
    GBARECOMP_DEFAULT_GAME_CONFIG="game.toml")

if(MSVC)
    target_link_libraries(TLARecomp PRIVATE
        gbarecomp_runtime gbarecomp_debug gbarecomp_gba gbarecomp_armv4t)
    target_link_options(TLARecomp PRIVATE /STACK:16777216)
else()
    target_compile_options(TLARecomp PRIVATE -ffunction-sections -fdata-sections)
    if(TLA_SPLIT_GAME_CODE)
        # Every engine function the game code might call has to be in the
        # exe even when the engine itself never calls it.
        target_link_libraries(TLARecomp PRIVATE
            "-Wl,--whole-archive"
            gbarecomp_runtime gbarecomp_debug gbarecomp_gba gbarecomp_armv4t
            "-Wl,--no-whole-archive")
    else()
        target_link_libraries(TLARecomp PRIVATE
            "-Wl,--start-group"
            gbarecomp_runtime gbarecomp_debug gbarecomp_gba gbarecomp_armv4t
            "-Wl,--end-group")
    endif()
    target_link_options(TLARecomp PRIVATE -Wl,--gc-sections)
    if(MINGW OR WIN32)
        target_link_options(TLARecomp PRIVATE -Wl,--stack,16777216)
    endif()
    if(TLA_IPO_SUPPORTED AND TLA_LTO_JOBS)
        # Overrides the plain -flto INTERPROCEDURAL_OPTIMIZATION put on this
        # link line: LTRANS across N processes.
        target_link_options(TLARecomp PRIVATE -flto=${TLA_LTO_JOBS})
    endif()
    # Incremental LTO (GCC 15+): the link keeps each finished piece in
    # <build>/lto-cache and later redoes only the pieces whose code changed.
    # Relative path: GCC's LTO job list is a makefile and must not carry the
    # checkout's path. One build per tree at a time (no locking on Windows).
    option(TLA_LTO_INCREMENTAL "Cache LTO link pieces between builds" ON)
    if(TLA_IPO_SUPPORTED AND TLA_LTO_INCREMENTAL AND
       CMAKE_CXX_COMPILER_ID STREQUAL "GNU" AND
       CMAKE_CXX_COMPILER_VERSION VERSION_GREATER_EQUAL 15)
        file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/lto-cache")
        target_link_options(TLARecomp PRIVATE
            -flto-partition=cache
            -flto-incremental=lto-cache)
    endif()
endif()

if(TLA_SPLIT_GAME_CODE AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
    # GSRecomp's ELF split: PIC table access avoids executable copy relocations.
    target_compile_options(TLARecomp PRIVATE -fPIC)
    set_target_properties(tla_game_objects PROPERTIES POSITION_INDEPENDENT_CODE ON)
    target_compile_options(tla_game_objects PRIVATE -fno-semantic-interposition)
    add_library(TLAGame SHARED $<TARGET_OBJECTS:tla_game_objects>)
    set_target_properties(TLAGame PROPERTIES
        INTERPROCEDURAL_OPTIMIZATION FALSE
        LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    target_link_options(TLAGame PRIVATE -Wl,-Bsymbolic)
    target_link_libraries(TLARecomp PRIVATE TLAGame)
    set_target_properties(TLARecomp PROPERTIES
        BUILD_WITH_INSTALL_RPATH ON INSTALL_RPATH "$ORIGIN")
elseif(TLA_SPLIT_GAME_CODE)
    # The tables the engine reads from the game code, by name: every
    # dispatch_table.cpp and expected.cpp defines them as
    # `extern "C" const ... NAME`.
    set(TLA_GAME_DEF "${CMAKE_BINARY_DIR}/TLAGame.def")
    set(TLA_GAME_DEF_TEXT "LIBRARY TLAGame.dll\nEXPORTS\n")
    set(TLA_GAME_TABLE_FILES ${TLA_GAME_SOURCES})
    list(FILTER TLA_GAME_TABLE_FILES INCLUDE REGEX "/(dispatch_table|expected)\\.cpp$")
    foreach(TLA_TABLE_FILE IN LISTS TLA_GAME_TABLE_FILES)
        file(STRINGS "${TLA_TABLE_FILE}" TLA_TABLE_LINES
            REGEX "^extern \"C\" const ")
        foreach(TLA_TABLE_LINE IN LISTS TLA_TABLE_LINES)
            if(TLA_TABLE_LINE MATCHES "([A-Za-z_][A-Za-z0-9_]*)(\\[\\])? =")
                string(APPEND TLA_GAME_DEF_TEXT "    ${CMAKE_MATCH_1} DATA\n")
            endif()
        endforeach()
    endforeach()
    file(WRITE "${TLA_GAME_DEF}.tmp" "${TLA_GAME_DEF_TEXT}")
    file(COPY_FILE "${TLA_GAME_DEF}.tmp" "${TLA_GAME_DEF}" ONLY_IF_DIFFERENT)

    # The exe imports those tables from TLAGame.dll through an import
    # library made from the .def alone, so the exe never waits on the DLL.
    set(TLA_GAME_IMPLIB "${CMAKE_BINARY_DIR}/libTLAGame_tables.a")
    find_program(TLA_DLLTOOL dlltool REQUIRED)
    add_custom_command(OUTPUT "${TLA_GAME_IMPLIB}"
        COMMAND "${TLA_DLLTOOL}" -d "${TLA_GAME_DEF}" -l "${TLA_GAME_IMPLIB}"
                -D TLAGame.dll
        DEPENDS "${TLA_GAME_DEF}"
        COMMENT "Import library for TLAGame.dll's tables")
    add_custom_target(tla_game_tables_implib DEPENDS "${TLA_GAME_IMPLIB}")
    add_dependencies(TLARecomp tla_game_tables_implib)
    target_link_libraries(TLARecomp PRIVATE "${TLA_GAME_IMPLIB}")

    # The game code: no LTO, compiled once and reused until the translated
    # code changes. Objects first, so the exe's export list can be read from
    # them before the exe links.
    # -Og and no debug information: Golden Sun Recompiled's measured choice
    # for the compile a player's PC waits for.
    add_library(TLAGame SHARED $<TARGET_OBJECTS:tla_game_objects>)
    set_target_properties(TLAGame PROPERTIES
        PREFIX ""
        INTERPROCEDURAL_OPTIMIZATION FALSE
        RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
    set(TLA_ENGINE_API_IMPLIB "${CMAKE_BINARY_DIR}/libTLARecomp_api.a")
    add_dependencies(TLAGame TLARecomp)
    target_link_libraries(TLAGame PRIVATE "${TLA_ENGINE_API_IMPLIB}")
    target_link_options(TLAGame PRIVATE "${TLA_GAME_DEF}")

    # What the exe exports: every symbol the game objects use that the
    # engine defines (cmake/gsr_engine_exports.cmake), worked out right
    # before the exe links. gcc-nm reads the engine's LTO objects.
    find_program(TLA_GCC_NM gcc-nm REQUIRED)
    set(TLA_ENGINE_EXPORTS "${CMAKE_BINARY_DIR}/TLARecomp_exports.def")
    set(TLA_ENGINE_EXPORTS_RSP "${CMAKE_BINARY_DIR}/TLARecomp_exports.rsp")
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/tla_game_objects.txt"
        CONTENT "$<JOIN:$<TARGET_OBJECTS:tla_game_objects>,\n>\n")
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/tla_engine_inputs.txt"
        CONTENT "$<TARGET_FILE:gbarecomp_runtime>\n$<TARGET_FILE:gbarecomp_debug>\n$<TARGET_FILE:gbarecomp_gba>\n$<TARGET_FILE:gbarecomp_armv4t>\n$<JOIN:$<TARGET_OBJECTS:TLARecomp>,\n>\n")
    add_dependencies(TLARecomp tla_game_objects)
    add_custom_command(TARGET TLARecomp PRE_LINK
        COMMAND "${CMAKE_COMMAND}"
            "-DNM=${TLA_GCC_NM}"
            "-DGAME_LIST=${CMAKE_BINARY_DIR}/tla_game_objects.txt"
            "-DENGINE_LIST=${CMAKE_BINARY_DIR}/tla_engine_inputs.txt"
            "-DOUT=${TLA_ENGINE_EXPORTS}"
            "-DOUT_RSP=${TLA_ENGINE_EXPORTS_RSP}"
            -P "${CMAKE_SOURCE_DIR}/cmake/gsr_engine_exports.cmake"
        # The DLL's import library comes from the same list: the one ld
        # writes with --out-implib under LTO leaves out exports the engine
        # itself never calls.
        COMMAND "${TLA_DLLTOOL}" -d "${TLA_ENGINE_EXPORTS}"
            -l "${TLA_ENGINE_API_IMPLIB}" -D TLAEngine.dll
        BYPRODUCTS "${TLA_ENGINE_EXPORTS}" "${TLA_ENGINE_EXPORTS_RSP}"
                   "${TLA_ENGINE_API_IMPLIB}"
        WORKING_DIRECTORY "${CMAKE_BINARY_DIR}"
        VERBATIM)
    target_link_options(TLARecomp PRIVATE "${TLA_ENGINE_EXPORTS}"
        "@${TLA_ENGINE_EXPORTS_RSP}")
endif()
